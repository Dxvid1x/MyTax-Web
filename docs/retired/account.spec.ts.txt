const user = {
  id: "11111111-1111-4111-8111-111111111111",
  email: "member@example.com",
};
type Database = {
  document: CloudDocument;
  revision: number;
  saves: CloudDocument[];
  fail?: boolean;
  otherUser?: boolean;
};
async function provider(page: Page, db: Database, signedIn = false) {
  let authenticated = signedIn;
  await page.route("**/api/ledger", async (route) => {
    const body =
      route.request().method() === "POST" ? route.request().postDataJSON() : {};
    const respond = (value: unknown, status = 200) =>
      route.fulfill({
        status,
        contentType: "application/json",
        body: JSON.stringify(value),
      });
    if (!body.action) return respond({ configured: true });
    if (body.action === "send-code")
      return respond(
        body.consent ? { ok: true } : { error: "ต้องยืนยันก่อน" },
        body.consent ? 200 : 400,
      );
    if (body.action === "verify-code") {
      authenticated = true;
      return respond({ user });
    }
    if (body.action === "session")
      return respond({ user: authenticated ? user : null });
    if (body.action === "logout") {
      authenticated = false;
      return respond({ ok: true });
    }
    if (!authenticated) return respond({ error: "โปรดเข้าสู่ระบบ" }, 401);
    if (body.action === "load")
      return respond({
        userId: db.otherUser ? "different-account" : user.id,
        record: db.revision
          ? { document: db.document, revision: db.revision }
          : null,
      });
    if (body.action === "save") {
      if (db.fail)
        return respond(
          { error: "บันทึกออนไลน์ไม่สำเร็จ ข้อมูลยังอยู่ในหน้านี้" },
          503,
        );
      if (body.revision !== db.revision)
        return respond(
          {
            error:
              "ข้อมูลเปลี่ยนจากอีกเครื่องแล้ว สำรองหน้านี้ก่อนโหลดข้อมูลล่าสุด",
          },
          409,
        );
      db.document = body.document;
      db.saves.push(body.document);
      db.revision++;
      return respond({ revision: db.revision, userId: user.id });
    }
    return respond({ error: "ไม่รองรับ" }, 400);
  });
}
const database = (): Database => ({
  document: emptyCloudDocument(),
  revision: 0,
  saves: [],
});
async function entry(page: Page, note: string, recurring = false) {
  await page.getByRole("button", { name: "จดเงินออก" }).click();
  await page.getByLabel("จำนวนเงิน").fill("80");
  await page.locator("#transaction-note").fill(note);
  if (recurring)
    await page.getByLabel("เป็นรายการประจำ", { exact: true }).check();
  await page.getByRole("button", { name: "เพิ่มรายการ", exact: true }).click();
}

test("intro crossfades into the running hero without a skip button, once per tab; honours Escape and reduced motion", async ({
  page,
}) => {
  await page.goto("/");
  const intro = page.getByRole("dialog", { name: "ภาพเปิด MyTax" });
  await expect(intro).toBeVisible();
  await expect(intro.getByRole("button")).toHaveCount(0);
  await expect(intro).toHaveAttribute("data-phase", "leaving");
  await expect
    .poll(() => intro.evaluate((el) => Number(getComputedStyle(el).opacity)))
    .toBeLessThan(0.95);
  expect(
    await intro.evaluate(
      (el) => getComputedStyle(el, "::backdrop").backgroundColor,
    ),
  ).toBe("rgba(0, 0, 0, 0)");
  expect(
    await page
      .locator(".opening-camera")
      .evaluate((el) => getComputedStyle(el).animationPlayState),
  ).toBe("running");
  await expect(intro).toHaveCount(0);
  await page.reload();
  await expect(intro).toHaveCount(0);
  await page.evaluate(() => sessionStorage.removeItem("mytax.intro.seen"));
  await page.reload();
  await expect(intro).toBeFocused();
  await page.keyboard.press("Escape");
  await expect(intro).toHaveCount(0);
  await page.evaluate(() => sessionStorage.removeItem("mytax.intro.seen"));
  await page.emulateMedia({ reducedMotion: "reduce" });
  await page.reload();
  await expect(intro).toHaveCount(0);
  await expect(
    page.getByRole("button", { name: "คำนวณภาษีของฉัน", exact: true }).first(),
  ).toBeVisible();
});
test("intro artwork fills both edges and keeps three bouncing coins clear of the logo across screen sizes", async ({
  page,
}) => {
  // Hold the opening timers while inspecting the same visual frame at each size.
  await page.clock.install();
  await page.clock.pauseAt(new Date(Date.now() + 1000));
  for (const width of [390, 1280, 1920]) {
    await page.setViewportSize({ width, height: 900 });
    for (const colorScheme of ["light", "dark"] as const) {
      await page.emulateMedia({ colorScheme });
      await page.goto("/");
      const intro = page.getByRole("dialog", { name: "ภาพเปิด MyTax" });
      await expect(intro).toBeVisible();
      await expect(intro.locator(".intro-bouncing-coin")).toHaveCount(3);
      await expect(intro.locator(".intro-chart-rise")).toHaveCount(2);
      await expect(intro.locator(".intro-chart-fall")).toHaveCount(2);
      await page.evaluate(async () => {
        await document.fonts.ready;
        for (const animation of document.getAnimations()) {
          if (animation.effect?.getTiming().iterations === Infinity) continue;
          animation.pause();
          animation.currentTime = 1600;
        }
      });
      const layout = await intro.evaluate((el) => {
        const rect = el.getBoundingClientRect();
        const brand = el
          .querySelector(".intro-brand-scene")!
          .getBoundingClientRect();
        const charts = [...el.querySelectorAll(".intro-chart")].map((chart) =>
          chart.getBoundingClientRect(),
        );
        const coins = [...el.querySelectorAll(".intro-bouncing-coin")].map(
          (coin) => coin.getBoundingClientRect(),
        );
        return {
          fullWidth: rect.left === 0 && rect.right === innerWidth,
          edgeCharts: charts[0].left === 0 && charts[1].right === innerWidth,
          noOverflow: document.documentElement.scrollWidth <= innerWidth,
          clearLogo: coins.every(
            (coin) =>
              coin.right <= brand.left ||
              coin.left >= brand.right ||
              coin.bottom <= brand.top ||
              coin.top >= brand.bottom,
          ),
        };
      });
      expect(layout).toEqual({
        fullWidth: true,
        edgeCharts: true,
        noOverflow: true,
        clearLogo: true,
      });
      await expect(intro.locator(".intro-chart-rise").first()).toHaveCSS(
        "stroke",
        "rgb(99, 223, 152)",
      );
      await expect(intro.locator(".intro-chart-fall").first()).toHaveCSS(
        "stroke",
        "rgb(255, 103, 135)",
      );
      await page.screenshot({
        path: `output/playwright/intro-artwork-${colorScheme}-${width}.png`,
      });
      await page.evaluate(() => sessionStorage.removeItem("mytax.intro.seen"));
    }
  }
});

test("email login uploads nothing until local migration is explicitly confirmed", async ({
  page,
}) => {
  const db = database();
  await provider(page, db);
  const saved = makeSaved(emptyPlan(), [
    createTransaction("expense", "80", today(), "อาหาร", "สำเนาในเครื่อง"),
  ]);
  await page.addInitScript(({ key, raw }) => localStorage.setItem(key, raw), {
    key: STORAGE_KEY,
    raw: JSON.stringify(saved),
  });
  await page.goto("/ledger");
  await expect(page.locator(".transaction")).toContainText("สำเนาในเครื่อง");
  await page.getByRole("button", { name: "ใช้ข้ามเครื่อง" }).click();
  await page.getByLabel("อีเมล", { exact: true }).fill(user.email);
  await page.getByRole("checkbox").check();
  await page.getByRole("button", { name: "ส่งรหัสเข้าอีเมล" }).click();
  await page.getByLabel("รหัสจากอีเมล").fill("123456");
  await page.getByRole("button", { name: "ยืนยันเข้าสู่ระบบ" }).click();
  await expect(page.getByRole("dialog")).toHaveCount(0);
  expect(db.saves).toHaveLength(0);
  await expect(page.locator(".transaction")).toHaveCount(0);
  await page.getByText("จัดการสมุดเงินออนไลน์", { exact: true }).click();
  await page
    .getByRole("button", { name: "นำสมุดเงินในเครื่องเข้าบัญชี" })
    .click();
  await expect(page.getByText(/เพิ่ม 1 รายการ/)).toBeVisible();
  await expect(
    page.getByRole("button", { name: "ยืนยันนำเข้าบัญชี" }),
  ).toBeDisabled();
  expect(db.saves).toHaveLength(0);
  await page.getByRole("checkbox").check();
  await page.getByRole("button", { name: "ยืนยันนำเข้าบัญชี" }).click();
  await expect(
    page.getByText("บันทึกออนไลน์แล้ว", { exact: true }),
  ).toBeVisible();
  expect(db.document.transactions).toHaveLength(1);
  expect(JSON.stringify(db.saves)).not.toMatch(
    /taxPlan|salary|access_token|refresh_token/,
  );
  const local = await page.evaluate(
    (key) => JSON.parse(localStorage.getItem(key)!),
    STORAGE_KEY,
  );
  expect(local.transactions).toHaveLength(1);
  await page.getByRole("button", { name: "ออกจากระบบ" }).click();
  await expect(page.locator(".transaction")).toContainText("สำเนาในเครื่อง");
  expect(db.document.transactions).toHaveLength(1);
});
test("two devices read the same ledger; recurring writes stay atomic and stale writes pause", async ({
  browser,
}) => {
  const db = database();
  const first = await browser.newContext(),
    second = await browser.newContext();
  try {
    const a = await first.newPage(),
      b = await second.newPage();
    await provider(a, db, true);
    await provider(b, db, true);
    await a.goto("/ledger");
    await expect(a.getByText("พร้อมจดออนไลน์", { exact: true })).toBeVisible();
    await entry(a, "ค่าอาหารประจำ", true);
    await expect(
      a.getByText("บันทึกออนไลน์แล้ว", { exact: true }),
    ).toBeVisible();
    expect(db.document.transactions).toHaveLength(1);
    expect(db.document.settings.recurring).toHaveLength(1);
    await b.goto("/ledger");
    await expect(b.locator(".transaction")).toContainText("ค่าอาหารประจำ");
    await entry(a, "เพิ่มจากเครื่องแรก");
    await expect(
      a.getByText("บันทึกออนไลน์แล้ว", { exact: true }),
    ).toBeVisible();
    await entry(b, "รายการยังไม่ซิงก์");
    await expect(
      b.getByText("พักการซิงก์ · ข้อมูลต่างกัน", { exact: true }),
    ).toBeVisible();
    expect(db.document.transactions).toHaveLength(2);
    await expect(
      b.locator(".transaction").filter({ hasText: "รายการยังไม่ซิงก์" }),
    ).toBeVisible();
    await expect(b.getByRole("button", { name: "ออกจากระบบ" })).toBeDisabled();
    await b.getByRole("button", { name: "ตรวจและโหลดข้อมูลล่าสุด" }).click();
    await expect(
      b.getByRole("button", { name: "โหลดข้อมูลล่าสุด", exact: true }),
    ).toBeDisabled();
    const downloaded = b.waitForEvent("download");
    await b.getByRole("button", { name: "ดาวน์โหลดสำเนาหน้านี้" }).click();
    await downloaded;
    await b
      .getByRole("button", { name: "โหลดข้อมูลล่าสุด", exact: true })
      .click();
    await expect(
      b.locator(".transaction").filter({ hasText: "เพิ่มจากเครื่องแรก" }),
    ).toBeVisible();
    await expect(
      b.locator(".transaction").filter({ hasText: "รายการยังไม่ซิงก์" }),
    ).toHaveCount(0);
    await b.goto("/ledger");
  } finally {
    await first.close();
    await second.close();
  }
});
test("failed writes remain visible, can retry and account mismatch never loads foreign data", async ({
  page,
}) => {
  const db = database();
  await provider(page, db, true);
  await page.goto("/ledger");
  await expect(page.getByText("พร้อมจดออนไลน์", { exact: true })).toBeVisible();
  db.fail = true;
  await entry(page, "เก็บไว้ก่อน");
  await expect(
    page.getByText("ยังบันทึกออนไลน์ไม่ได้", { exact: true }),
  ).toBeVisible();
  await expect(page.locator(".transaction")).toContainText("เก็บไว้ก่อน");
  expect(db.saves).toHaveLength(0);
  db.fail = false;
  await page.getByRole("button", { name: "ลองบันทึกอีกครั้ง" }).click();
  await expect(
    page.getByText("บันทึกออนไลน์แล้ว", { exact: true }),
  ).toBeVisible();
  db.otherUser = true;
  await page.getByText("จัดการสมุดเงินออนไลน์", { exact: true }).click();
  await page
    .getByRole("button", { name: "โหลดข้อมูลล่าสุด", exact: true })
    .click();
  await expect(
    page.getByText("พักการซิงก์ · ข้อมูลต่างกัน", { exact: true }),
  ).toBeVisible();
  await expect(page.locator(".transaction")).toContainText("เก็บไว้ก่อน");
});
test("ledger storage choices are readable, keyboard accessible and fit both themes", async ({
  page,
}) => {
  for (const theme of ["light", "dark"]) {
    await page.addInitScript(
      (value) => localStorage.setItem("mytax.theme", value),
      theme,
    );
    for (const width of [390, 768, 1280]) {
      await page.setViewportSize({ width, height: 900 });
      await page.goto("/ledger");
      await expect(
        page.getByRole("button", { name: "ใช้ข้ามเครื่อง" }),
      ).toBeVisible();
      expect(
        await page.evaluate(
          () => document.documentElement.scrollWidth <= innerWidth,
        ),
      ).toBe(true);
      await page.getByRole("button", { name: "ใช้ข้ามเครื่อง" }).focus();

      await expect(page.getByRole("dialog")).toBeVisible();

      await page.screenshot({
        path: `output/playwright/account-${theme}-${width}.png`,
      });
      await page.keyboard.press("Escape");
      expect((await new AxeBuilder({ page }).analyze()).violations).toEqual([]);
    }
  }
});
